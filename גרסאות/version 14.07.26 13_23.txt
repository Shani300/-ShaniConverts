        private void SetStartup()
        {
            try
            {
                RegistryKey rk = Registry.CurrentUser.OpenSubKey("SOFTWARE\\Microsoft\\Windows\\CurrentVersion\\Run", true);
                rk.SetValue("ShaniConverts", Application.ExecutablePath);
            }
            catch (Exception ex)
            {
                System.Diagnostics.Debug.WriteLine("Registry error: " + ex.Message);
            }
        }

        // פונקציית עזר לניקוי ה-Buffer ממקשים לחוצים (פותר את בעיית ה-a הנדבקת)
        private void ClearHardwareModifierKeys()
        {
            keybd_event(VK_CONTROL, 0, KEYEVENTF_KEYUP, 0);
            keybd_event(VK_SHIFT, 0, KEYEVENTF_KEYUP, 0);
            Thread.Sleep(50); // השהיה קלה לעדכון תור הקלטים במערכת
        }

        private void OnKeyDown(object sender, KeyEventArgs e)
        {
            if (e.Control && e.Shift && e.KeyCode == Keys.Z)
            {
                e.Handled = true;
                this.BeginInvoke((MethodInvoker)delegate { ConvertAllText(); });
            }
            else if (e.Control && e.Shift && e.KeyCode == Keys.A)
            {
                e.Handled = true;
                this.BeginInvoke((MethodInvoker)delegate { ConvertSelectedText(); });
            }
        }

        private void OnMouseDown(object sender, MouseEventExtArgs e)
        {
            if (e.Button == MouseButtons.Middle)
            {
                e.Handled = true;
                this.BeginInvoke((MethodInvoker)delegate { ExecuteConversionCore(); });
            }
        }

        // פונקציה עבור Ctrl+Shift+Z (בחירה אוטומטית והמרה)
        private void ConvertAllText()
        {
            try
            {
                ClearHardwareModifierKeys();

                SendKeys.SendWait("^a"); // בחירת הכל
                Thread.Sleep(80);        // השהיה קצרצרה במיוחד כדי ש-VS יספיק לסמן

                ExecuteConversionCore();
            }
            catch (Exception ex)
            {
                System.Diagnostics.Debug.WriteLine(ex.Message);
            }
        }

        // פונקציה עבור Ctrl+Shift+A (המרה של טקסט מסומן קיים)
        private void ConvertSelectedText()
        {
            try
            {
                ClearHardwareModifierKeys();
                ExecuteConversionCore();
            }
            catch (Exception ex)
            {
                System.Diagnostics.Debug.WriteLine(ex.Message);
            }
        }

        // ליבת הביצוע האולטרה-מהירה (ללא חסימת ממשק)
        private void ExecuteConversionCore()
        {
            try
            {
                // שמירת הלוח הישן
                string oldClipboard = Clipboard.ContainsText() ? Clipboard.GetText() : "";
                Clipboard.Clear();

                // העתקה באמצעות Ctrl+C (מהיר וטבעי יותר עבור VS / VS Code)
                SendKeys.SendWait("^c");
                Thread.Sleep(70); // השהיה מינימלית הנדרשת להעתקה למערכת

                if (!Clipboard.ContainsText()) return;

                string text = Clipboard.GetText();
                string converted = ProcessText(text);

                Clipboard.SetText(converted);

                // הדבקה באמצעות Ctrl+V
                SendKeys.SendWait("^v");

                // פתרון הדיליי: שחזור ה-Clipboard הישן מתבצע ב-Thread נפרד לחלוטין ברקע!
                // זה מונע מהתוכנה לקפוא ומאפשר לך להמשיך להקליד קוד מיד ללא שום השהיה.
                if (!string.IsNullOrEmpty(oldClipboard))
                {
                    System.Threading.Tasks.Task.Run(() =>
                    {
                        Thread.Sleep(400); // ממתין ברקע בשקט שסביבת הפיתוח תסיים לקרוא את ה-Ctrl+V

                        if (this.IsHandleCreated)
                        {
                            this.BeginInvoke((MethodInvoker)delegate
                            {
                                try
                                {
                                    Clipboard.SetText(oldClipboard); // מחזיר את הלוח הישן בחזרה למערכת
                                }
                                catch { /* הגנה מפני קריסה אם הלוח תפוס */ }
                            });
                        }
                    });
                }
            }
            catch (Exception ex)
            {
                System.Diagnostics.Debug.WriteLine(ex.Message);
            }
        }

        private string ProcessText(string input)
        {
            bool hasHebrew = input.Any(c => c >= 'א' && c <= 'ת');
            var mapping = hasHebrew ? _heToEn : _enToHe;

            char[] result = new char[input.Length];
            for (int i = 0; i < input.Length; i++)
            {
                result[i] = mapping.ContainsKey(input[i]) ? mapping[input[i]] : input[i];
            }
            return new string(result);
        }

        private static Dictionary<char, char> CreateMapping(string from, string to)
        {
            var dict = new Dictionary<char, char>();
            for (int i = 0; i < from.Length; i++)
            {
                if (i < to.Length) dict[from[i]] = to[i];
            }
            return dict;
        }
